# ============================================================
# SISTEMA ESPECIALISTA: MOTOR DE INFERÊNCIA FORWARD CHAINING
# ============================================================

class BaseDeConhecimento:
    def __init__(self):
        # Memória de Trabalho: Armazena os fatos válidos na sessão
        self.fatos = set()
        # Regras de Produção: Lista de tuplas (conjunto_de_premissas, conclusao_ou_acao)
        # Notação Lógica: Premissas (P1 ∧ P2 ∧ ...) ⟹ Consequente (Q)
        self.regras = []

    def adicionar_fato(self, fato: str):
        """Insere um novo fato confirmado na memória de trabalho."""
        self.fatos.add(fato)

    def adicionar_regra(self, premissas: list, consequente: str):
        """Define uma regra no formato: SE (p1 ∧ p2 ∧ ...) ⟹ ENTÃO (q)."""
        self.regras.append((set(premissas), consequente))


class MotorDeInferênciaForward:
    def __init__(self, base_conhecimento: BaseDeConhecimento):
        self.kb = base_conhecimento
        self.historico_disparos = []

    def executar(self):
        """Ciclo Match-Resolve-Act até saturar todas as deduções possíveis."""
        novos_fatos = True
        regras_aplicadas = set()

        print("=======================================================")
        print("  INICIANDO PROCESSO DE INFERÊNCIA (FORWARD CHAINING)  ")
        print("=======================================================")
        
        while novos_fatos:
            novos_fatos = False
            for idx, (premissas, consequente) in enumerate(self.kb.regras):
                if idx in regras_aplicadas:
                    continue
                
                # Match: Verifica se todas as premissas são um subconjunto dos fatos atuais
                # Lógica: Premissas ⊆ Fatos ⟹ Dispara Regra (Modus Ponens)
                if premissas.issubset(self.kb.fatos):
                    premissas_str = " ∧ ".join(list(premissas))
                    print(f"-> [DISPARO]: ({premissas_str}) ⟹ '{consequente}'")
                    self.kb.adicionar_fato(consequente)
                    self.historico_disparos.append((list(premissas), consequente))
                    regras_aplicadas.add(idx)
                    novos_fatos = True
                    break # Resolve & Act: reinicia o ciclo com a memória de trabalho atualizada

        print("\n-------------------------------------------------------")
        print("  MEMÓRIA DE TRABALHO FINAL (FATOS CONHECIDOS/INFERIDOS) ")
        print("-------------------------------------------------------")
        for fato in sorted(self.kb.fatos):
            print(f" • {fato}")
        print("=======================================================\n")

if __name__ == "__main__":

    # 2. Definição das Regras do Especialista de Suporte (Lógica de Produção)
    # R1: sem_internet ∧ ping_roteador_ok ⟹ problema_provedor
    kb.adicionar_regra(["sem_internet", "ping_roteador_ok"], "problema_provedor")
    
    # R2: sem_internet ∧ ping_roteador_falhou ⟹ problema_rede_local

# ============================================================
# RESOLUÇÃO DO PASSO 3: MOTOR DE INFERÊNCIA INTERATIVO
# ============================================================

class MotorInterativoForward:
    def __init__(self, base_conhecimento: BaseDeConhecimento):
        self.kb = base_conhecimento
        self.perguntados = set()

    def _obter_fatos_inferiveis(self):
        """Identifica todos os fatos que podem ser deduzidos como conclusões de regras."""
        return {consequente for _, consequente in self.kb.regras}

    def interagir_e_executar(self):
        conclusoes_possiveis = self._obter_fatos_inferiveis()
        novos_fatos = True
        regras_aplicadas = set()

        print("\n--- INICIANDO DIAGNÓSTICO INTERATIVO COM O USUÁRIO ---")

        while novos_fatos:
            novos_fatos = False
            for idx, (premissas, consequente) in enumerate(self.kb.regras):
                if idx in regras_aplicadas:
                    continue

                # Pergunta ao usuário sobre premissas primitivas que ainda não conhecemos
                for premissa in premissas:
                    if premissa not in self.kb.fatos and premissa not in self.perguntados:
                        # Se não pode ser deduzido por outra regra, é um fato observável primitivo
                        if premissa not in conclusoes_possiveis:
                            resp = input(f"-> O fato/sintoma '{premissa}' está presente? (s/n): ").strip().lower()
                            self.perguntados.add(premissa)
                            if resp == 's' or resp == 'sim':
                                self.kb.adicionar_fato(premissa)

                # Verifica se a regra agora dá Match
                if premissas.issubset(self.kb.fatos):
                    print(f"[REGRA DISPARADA]: SE {' ∧ '.join(premissas)} ⟹ '{consequente}'")
                    self.kb.adicionar_fato(consequente)
                    regras_aplicadas.add(idx)
                    novos_fatos = True
                    break

        print("\n--- DIAGNÓSTICO FINAL CONCLUÍDO ---")
        for fato in sorted(self.kb.fatos):
            if fato.startswith("triagem:"):
                print(f"🎯 DECISÃO FINAL: {fato}")

# ============================================================
# RESOLUÇÃO DO PASSO 4: MÓDULO DE EXPLICAÇÃO ("COMO?")
# ============================================================

class MotorComExplicacao(MotorDeInferênciaForward):
    def __init__(self, base_conhecimento: BaseDeConhecimento):
        super().__init__(base_conhecimento)
        # Mapeia: fato_gerado -> conjunto_de_premissas_que_o_geraram
        self.grafo_justificativas = {}

    def executar(self):
        novos_fatos = True
        regras_aplicadas = set()

        while novos_fatos:
            novos_fatos = False
            for idx, (premissas, consequente) in enumerate(self.kb.regras):
                if idx in regras_aplicadas:
                    continue
                if premissas.issubset(self.kb.fatos):
                    self.kb.adicionar_fato(consequente)
                    self.grafo_justificativas[consequente] = list(premissas)
                    regras_aplicadas.add(idx)
                    novos_fatos = True
                    break

    def explicar_como(self, fato_alvo: str, nivel: int = 0):
        """Reconstrói recursivamente a árvore lógica que justificou o fato."""
        ident = "  " * nivel
        if fato_alvo in self.grafo_justificativas:
            premissas = self.grafo_justificativas[fato_alvo]
            print(f"{ident}└─ '{fato_alvo}' foi DEDUZIDO pela regra: ({' ∧ '.join(premissas)} ⟹ '{fato_alvo}')")
            for p in premissas:
                self.explicar_como(p, nivel + 1)
        else:
            print(f"{ident}├─ '{fato_alvo}' foi informado como FATO INICIAL ou confirmado pelo usuário.")

Implementação completa do algoritmo de Backward Chaining guiado por hipóteses com busca recursiva:

# ============================================================
# RESOLUÇÃO DO PASSO 5: MOTOR BACKWARD CHAINING RECURSIVO
# ============================================================

class MotorDeInferênciaBackward:
    def __init__(self, base_conhecimento: BaseDeConhecimento):
        self.kb = base_conhecimento
        self.perguntados = set()

    def provar_meta(self, meta: str, nivel: int = 0) -> bool:
        """Tenta provar recursivamente uma hipótese/meta."""
        ident = "  " * nivel
        print(f"{ident}🔍 Investigando meta: '{meta}'...")

        # 1. Caso Base: O fato já está confirmado na memória de trabalho
        if meta in self.kb.fatos:
            print(f"{ident}✅ Fato '{meta}' já é conhecido e confirmado na memória.")
            return True

        # 2. Busca regras que tenham a meta no consequente (P ⟹ meta)
        regras_candidatas = [r for r in self.kb.regras if r[1] == meta]

        for premissas, _ in regras_candidatas:
            print(f"{ident}→ Avaliando regra: {' ∧ '.join(premissas)} ⟹ '{meta}'")
            # Todas as premissas da regra precisam ser provadas (recursão)
            todas_premissas_ok = True
            for p in premissas:
                if not self.provar_meta(p, nivel + 1):
                    todas_premissas_ok = False
                    break # Se uma premissa falhar, descarta essa regra e tenta a próxima

            if todas_premissas_ok:
                print(f"{ident}🎉 Meta '{meta}' foi PROVADA com sucesso!")
                self.kb.adicionar_fato(meta)
                return True

        # 3. Se não há regras para provar a meta, pergunta diretamente ao usuário (fato primitivo)
        if not regras_candidatas and meta not in self.perguntados:
            resp = input(f"\n[PERGUNTA]: O fato '{meta}' é verdadeiro? (s/n): ").strip().lower()
            self.perguntados.add(meta)
            if resp == 's' or resp == 'sim':
                self.kb.adicionar_fato(meta)
                return True

        print(f"{ident}❌ Não foi possível provar a meta: '{meta}'.")
        return False

# ==========================================
if __name__ == "__main__":
